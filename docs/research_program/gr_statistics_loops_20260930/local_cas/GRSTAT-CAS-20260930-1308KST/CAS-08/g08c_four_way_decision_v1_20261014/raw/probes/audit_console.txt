def G08C.fourWayReport.{u} : {Θ : Type u} → {C CG : Set Θ} → Cert C → Cert CG → Report :=
fun {Θ} {C CG} x x_1 =>
  match x, x_1 with
  | Cert.excluded h, x => Report.modelIncompatible
  | x, Cert.feasible θ h => Report.notExcluded
  | Cert.feasible θ h, Cert.excluded h_1 => Report.geodesicityRejected
  | x, x_2 => Report.computationallyUnresolved
def G08C.confSet.{u} : {Θ : Type u} → Set Θ → (Θ → ℝ) → ℝ → Set Θ :=
fun {Θ} Ω Q q => {θ | θ ∈ Ω ∧ Q θ ≤ q}
def G08C.nullConfSet.{u, v} : {Θ : Type u} → {X : Type v} → Set Θ → (Θ → ℝ) → ℝ → (Θ → X) → Set X → Set Θ :=
fun {Θ} {X} Ω Q q xmap G => {θ | θ ∈ confSet Ω Q q ∧ xmap θ ∈ G}
def G08C.geodesicNull : Set FirstJet :=
