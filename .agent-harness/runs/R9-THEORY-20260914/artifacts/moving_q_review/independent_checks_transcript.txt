    q=mq.normalize_q(rng.normal(size=(3,3))); qp=mq.normalize_q(q+10**rng.uniform(-8,0)*rng.normal(size=(3,3)))
    et=float(rng.choice([0.0,.11,.55,.93,.999999,1.0])); ep=float(rng.choice([0.0,.21,.67,.97,.9999,1.0]))
    f=mq.fibre(q,rng.normal(size=3),et); g=mq.fibre(qp,rng.normal(size=3),ep)
    h=np.linalg.norm(q-qp); b=np.linalg.norm(f['v']-g['v']); eps=h+b

    # own 4096-direction lower witnesses and the candidate upper bounds
    x,_=mq.sample_points(f,rng,2048); y,_=mq.sample_points(g,rng,2048)
    w=max(mq.distances_to_fibre(x,g).max(),mq.distances_to_fibre(y,f).max())
    global_bound=2*mq.A*eps+np.sqrt(2*mq.A*eps)
    sep=D+np.sqrt(2*D)+np.sqrt(2)*min(1,mq.K*h)
    delta=1-max(et,ep)
    interior=mq.A*(2+np.sqrt((1-delta)/delta))*eps if delta>0 else None
    worst['global_ratio']=max(worst['global_ratio'],w/global_bound if global_bound else 0)
    worst['separated_ratio']=max(worst['separated_ratio'],w/sep if sep else 0)
    if interior: worst['interior_ratio']=max(worst['interior_ratio'],w/interior)
    proj=np.linalg.norm(f['p']-g['p'],2)
    worst['projection_ratio']=max(worst['projection_ratio'],proj/min(1,mq.K*h) if h else 0)
    centre=np.linalg.norm(f['c']-g['c'])
    worst['centre_ratio']=max(worst['centre_ratio'],centre/D if D else 0)
assert all(v <= 1+1e-10 for v in worst.values()), worst
